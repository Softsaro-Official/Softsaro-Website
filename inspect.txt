import { useEffect, useRef, useState } from "react";

/** Tries each src in order. Also checks on mount, because a 404 can fire before React hydrates. */
export default function WxPhoto({ srcs, alt, className, label }: { srcs: string[]; alt: string; className?: string; label?: string }) {
  const [i, setI] = useState(0);
  const ref = useRef<HTMLImageElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (el && el.complete && el.naturalWidth === 0) setI((n) => n + 1);
  }, [i]);

  if (i >= srcs.length) return <div className={"wx-empty " + (className || "")} role="img" aria-label={alt} />;
  return (
    // eslint-disable-next-line @next/next/no-img-element
    <img ref={ref} className={className} src={srcs[i]} alt={alt} loading="lazy" decoding="async" data-cursor-label={label} onError={() => setI((n) => n + 1)} />
  );
}

/** Masked word split (same technique as the other pages). Restores original markup on cleanup. */
function splitWords(el: HTMLElement) {
  const orig = el.innerHTML;
  const frag = document.createDocumentFragment();
  const wrap = (node: Node) => {
    const w = document.createElement("span");
    w.className = "wx-w";
    const i = document.createElement("span");
    i.className = "wx-i";
    i.appendChild(node);
    w.appendChild(i);
    return w;
  };
  Array.from(el.childNodes).forEach((n) => {
    if (n.nodeType === 3) {
      (n.textContent || "").split(/(\s+)/).forEach((p) => {
        if (!p) return;
        if (/^\s+$/.test(p)) frag.appendChild(document.createTextNode(" "));
        else frag.appendChild(wrap(document.createTextNode(p)));
      });
    } else if ((n as HTMLElement).tagName === "BR") frag.appendChild(n.cloneNode());
    else frag.appendChild(wrap(n.cloneNode(true)));
  });
  el.textContent = "";
  el.appendChild(frag);
  return { words: Array.from(el.querySelectorAll<HTMLElement>(".wx-i")), restore: () => { el.innerHTML = orig; } };
}
