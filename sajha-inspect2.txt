type Project = [string, string, string, string, string];

const projects: Project[] = D.pj;
type PageProps = { params: Promise<{ slug: string }> };

function findProject(slug: string) {
  return projects.find((project) => project[4] === slug);
}

export function generateStaticParams() {
  return projects.map((project) => ({ slug: project[4] }));
}

export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
  const { slug } = await params;
  const project = findProject(slug);

  if (!project) {
    return { title: "Project not found | Softsaro" };
  }

  return {
    title: `${project[0]} | Softsaro Portfolio`,
    description: project[2],
    alternates: { canonical: `/portfolio/${project[4]}` },
    openGraph: {
      title: `${project[0]} | Softsaro Portfolio`,
      description: project[2],
      images: [project[3]],
    },
  };
}
