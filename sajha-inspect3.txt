"use client";

import { Fragment, type ReactNode, type ElementType } from "react";
import Image from "next/image";
import { motion, useReducedMotion } from "framer-motion";
import type { SajhaImg } from "@/lib/sajha-data";
import s from "./sajha.module.css";

export const EASE = [0.16, 1, 0.3, 1] as const;
const VIEW = { once: true, margin: "-8% 0px" } as const;

type WordsProps = { text: string; as?: ElementType; className?: string; id?: string; inView?: boolean; delay?: number; stagger?: number };

/** Word-by-word masked heading reveal. */
export function Words({ text, as: Tag = "h2", className, id, inView = true, delay = 0, stagger = 0.06 }: WordsProps) {
  const rm = useReducedMotion();
  const words = text.split(" ");
  const trigger = inView ? { whileInView: { y: 0, rotate: 0 }, viewport: VIEW } : { animate: { y: 0, rotate: 0 } };
  return (
    <Tag className={className} id={id} aria-label={text}>
      {words.map((w, i) => (
        <Fragment key={i}>
          <span className={s.mask} aria-hidden="true">
            <motion.span className={s.maskIn} initial={rm ? false : { y: "112%", rotate: 4 }} {...trigger} transition={{ duration: 1.05, ease: EASE, delay: delay + i * stagger }}>{w}</motion.span>
          </span>
          {i < words.length - 1 ? " " : null}
        </Fragment>
      ))}
    </Tag>
  );
}

export function Rise({ children, className, delay = 0, y = 34 }: { children: ReactNode; className?: string; delay?: number; y?: number }) {
  const rm = useReducedMotion();
  return (
    <motion.div className={className} initial={rm ? false : { opacity: 0, y }} whileInView={{ opacity: 1, y: 0 }} viewport={VIEW} transition={{ duration: 1, ease: EASE, delay }}>
      {children}
    </motion.div>
  );
}

const container = (gap: number, delay: number) => ({ hidden: {}, show: { transition: { staggerChildren: gap, delayChildren: delay } } });
const item = { hidden: { opacity: 0, y: 30 }, show: { opacity: 1, y: 0, transition: { duration: 0.9, ease: EASE } } };

type StaggerProps = { children: ReactNode; className?: string; gap?: number; delay?: number };

export function Stagger({ children, className, gap = 0.1, delay = 0 }: StaggerProps) {
  const rm = useReducedMotion();
  return <motion.div className={className} variants={container(gap, delay)} initial={rm ? "show" : "hidden"} whileInView="show" viewport={VIEW}>{children}</motion.div>;
}
export function StaggerItem({ children, className }: { children: ReactNode; className?: string }) {
  return <motion.div className={className} variants={item}>{children}</motion.div>;
}
export function StaggerUl({ children, className, gap = 0.09, delay = 0 }: StaggerProps) {
  const rm = useReducedMotion();
  return <motion.ul className={className} variants={container(gap, delay)} initial={rm ? "show" : "hidden"} whileInView="show" viewport={VIEW}>{children}</motion.ul>;
}
export function StaggerLi({ children, className, node }: { children: ReactNode; className?: string; node?: boolean }) {
  return <motion.li className={className} variants={item} {...(node ? { "data-node": "" } : {})}>{children}</motion.li>;
}

type ClipProps = { img: SajhaImg; alt: string; sizes: string; priority?: boolean; caption?: ReactNode; className?: string; ratio?: string };

/** Clip-path reveal + inner scale + (GSAP) parallax. */
export function ClipImage({ img, alt, sizes, priority, caption, className = "", ratio }: ClipProps) {
  const rm = useReducedMotion();
  return (
    <figure className={`${s.fig} ${className}`}>
      <motion.div className={s.frame} data-frame style={{ aspectRatio: ratio ?? `${img.w} / ${img.h}` }}
        initial={rm ? false : { clipPath: "inset(0 0 100% 0)" }} whileInView={{ clipPath: "inset(0 0 0% 0)" }} viewport={VIEW} transition={{ duration: 1.3, ease: [0.76, 0, 0.24, 1] }}>
        <motion.div className={s.scaler} initial={rm ? false : { scale: 1.25 }} whileInView={{ scale: 1 }} viewport={VIEW} transition={{ duration: 1.8, ease: EASE }}>
          <div className={s.par} data-par>
            <Image src={img.src} alt={alt} fill sizes={sizes} priority={priority} placeholder="blur" blurDataURL={img.blur} className={s.img} />
          </div>
        </motion.div>
      </motion.div>
      {caption ? <figcaption className={s.cap}>{caption}</figcaption> : null}
    </figure>
  );
}

import { useEffect, useState } from "react";
import { AnimatePresence, motion, useMotionValueEvent, useScroll, useSpring } from "framer-motion";
import { EASE } from "./Fx";
import s from "./sajha.module.css";

const LINKS: [string, string][] = [
  ["overview", "Overview"], ["challenge", "Challenge"], ["solution", "Solution"], ["features", "Features"],
  ["mobile", "Mobile"], ["process", "Process"], ["gallery", "Gallery"], ["results", "Results"],
];

/** Scroll progress bar + floating section dock. */
export default function SajhaNav() {
  const { scrollY, scrollYProgress } = useScroll();
  const progress = useSpring(scrollYProgress, { stiffness: 120, damping: 28, mass: 0.3 });
  const [show, setShow] = useState(false);
  const [active, setActive] = useState("");
  useMotionValueEvent(scrollY, "change", (v) => setShow(v > 520));

  useEffect(() => {
    const els = LINKS.map(([id]) => document.getElementById(id)).filter(Boolean) as HTMLElement[];
    const io = new IntersectionObserver((entries) => entries.forEach((e) => { if (e.isIntersecting) setActive(e.target.id); }), { rootMargin: "-45% 0px -50% 0px" });
    els.forEach((el) => io.observe(el));
    return () => io.disconnect();
  }, []);

  return (
    <>
      <motion.div className={s.topbar} style={{ scaleX: progress }} aria-hidden="true" />
      <AnimatePresence>
        {show && (
          <motion.nav className={s.dock} aria-label="Case study sections" initial={{ y: 48, opacity: 0 }} animate={{ y: 0, opacity: 1 }} exit={{ y: 48, opacity: 0 }} transition={{ duration: 0.6, ease: EASE }}>
            {LINKS.map(([id, label]) => (
              <a key={id} href={`#${id}`} className={`${s.dl}${active === id ? ` ${s.dlOn}` : ""}`}>
                {active === id && <motion.span layoutId="sj-dock-pill" className={s.dockPill} transition={{ type: "spring", stiffness: 380, damping: 34 }} />}
                <span>{label}</span>
              </a>
            ))}
          </motion.nav>
        )}
      </AnimatePresence>
    </>
  );
}

"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import Image from "next/image";
import { AnimatePresence, motion, useReducedMotion, type Variants } from "framer-motion";
import { ArrowLeft, ArrowRight, Expand, X } from "lucide-react";
import type { SajhaImg } from "@/lib/sajha-data";
import { EASE } from "./Fx";
import s from "./sajha.module.css";

type Item = { img: SajhaImg; caption: string };
const pad = (n: number) => String(n).padStart(2, "0");
const slide: Variants = {
  enter: (d: number) => ({ x: d * 90, opacity: 0, scale: 0.96 }),
  center: { x: 0, opacity: 1, scale: 1 },
  exit: (d: number) => ({ x: -d * 90, opacity: 0, scale: 0.96 }),
};

export default function SajhaGallery({ items }: { items: Item[] }) {
  const rm = useReducedMotion();
  const [open, setOpen] = useState<number | null>(null);
  const [dir, setDir] = useState(1);
  const opener = useRef<HTMLElement | null>(null);
  const closeBtn = useRef<HTMLButtonElement>(null);
  const isOpen = open !== null;

  const go = useCallback((n: number) => {
    setDir(n > 0 ? 1 : -1);
    setOpen((i) => (i === null ? i : (i + n + items.length) % items.length));
  }, [items.length]);
  const close = useCallback(() => { setOpen(null); opener.current?.focus(); }, []);

  useEffect(() => {
    if (!isOpen) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape") close();
      if (e.key === "ArrowRight") go(1);
      if (e.key === "ArrowLeft") go(-1);
    };
    document.addEventListener("keydown", onKey);
    const prev = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    closeBtn.current?.focus();
    return () => { document.removeEventListener("keydown", onKey); document.body.style.overflow = prev; };
  }, [isOpen, go, close]);

  const t = { duration: rm ? 0 : 0.5, ease: EASE };
  return (
    <>
      <div className={s.masonry}>
        {items.map((it, i) => (
          <motion.button type="button" key={it.img.src} className={s.shot} data-cur="view" data-label="View" aria-label={`Open screenshot ${i + 1}: ${it.caption}`}
            initial={rm ? false : { opacity: 0, y: 44 }} whileInView={{ opacity: 1, y: 0 }} viewport={{ once: true, margin: "-6% 0px" }} transition={{ duration: 0.9, ease: EASE, delay: (i % 4) * 0.08 }}
            onClick={(e) => { opener.current = e.currentTarget; setDir(1); setOpen(i); }}>
            <span className={s.shotImg} style={{ aspectRatio: `${it.img.w} / ${it.img.h}` }}>
              <Image src={it.img.src} alt="" fill sizes="(min-width:1024px) 25vw, (min-width:640px) 33vw, 50vw" placeholder="blur" blurDataURL={it.img.blur} className={s.img} />
              <span className={s.shotIcon}><Expand size={16} /></span>
            </span>
            <span className={s.shotMeta}><em>{pad(i + 1)}</em><b>{it.caption}</b></span>
          </motion.button>
        ))}
      </div>

      <AnimatePresence>
        {open !== null && (
          <motion.div className={s.lb} role="dialog" aria-modal="true" aria-label="Project screenshots" initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} transition={{ duration: rm ? 0 : 0.35 }} onClick={close}>
            <div className={s.lbTop} onClick={(e) => e.stopPropagation()}>
              <span>{pad(open + 1)} / {pad(items.length)}</span>
              <button type="button" ref={closeBtn} className={s.lbBtn} onClick={close} aria-label="Close gallery"><X size={20} /></button>
            </div>
            <div className={s.lbStage} onClick={(e) => e.stopPropagation()}>
              <button type="button" className={`${s.lbNav} ${s.lbPrev}`} onClick={() => go(-1)} aria-label="Previous screenshot"><ArrowLeft size={20} /></button>
              <AnimatePresence mode="wait" custom={dir} initial={false}>
                <motion.div key={open} className={s.lbSlide} custom={dir} variants={slide} initial="enter" animate="center" exit="exit" transition={t}
                  drag="x" dragConstraints={{ left: 0, right: 0 }} dragElastic={0.25}
                  onDragEnd={(_, info) => { if (info.offset.x < -80) go(1); else if (info.offset.x > 80) go(-1); }}>
                  <Image src={items[open].img.src} alt={items[open].caption} width={items[open].img.w} height={items[open].img.h} sizes="90vw" className={s.lbImg} draggable={false} placeholder="blur" blurDataURL={items[open].img.blur} />
                </motion.div>
              </AnimatePresence>
              <button type="button" className={`${s.lbNav} ${s.lbNext}`} onClick={() => go(1)} aria-label="Next screenshot"><ArrowRight size={20} /></button>
            </div>
            <div className={s.lbBottom} onClick={(e) => e.stopPropagation()}>
              <p>{items[open].caption}</p>
              <div className={s.lbThumbs}>
                {items.map((it, i) => (
                  <button type="button" key={it.img.src} className={`${s.lbThumb}${i === open ? ` ${s.lbThumbOn}` : ""}`} aria-label={`Show screenshot ${i + 1}`} aria-current={i === open}
                    onClick={() => { setDir(i > open ? 1 : -1); setOpen(i); }}>
                    <Image src={it.img.src} alt="" fill sizes="80px" className={s.img} />
                  </button>
                ))}
              </div>
            </div>
          </motion.div>
        )}
      </AnimatePresence>
    </>
  );
}

import gsap from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";

gsap.registerPlugin(ScrollTrigger, useGSAP);

const SPRING = "elastic.out(1,.5)";

/** GSAP layer: parallax, drifting backgrounds, timeline, magnetic + tilt. Framer Motion handles entrances. */
export default function SajhaMotion({ children }: { children: ReactNode }) {
  const root = useRef<HTMLDivElement>(null);

  useGSAP(() => {
    const scope = root.current!;
    const mm = gsap.matchMedia(scope);

    mm.add({ motion: "(prefers-reduced-motion: no-preference)", fine: "(hover: hover) and (pointer: fine)", wide: "(min-width: 761px)" }, (ctx) => {
      const { motion, fine, wide } = ctx.conditions as { motion: boolean; fine: boolean; wide: boolean };
      if (!motion) return;
      const cleanups: Array<() => void> = [];
      const all = <T extends HTMLElement>(sel: string) => gsap.utils.toArray<T>(sel, scope);

      // image parallax
      all("[data-par]").forEach((el) => {
        gsap.fromTo(el, { yPercent: -6 }, { yPercent: 6, ease: "none", scrollTrigger: { trigger: el.closest("[data-frame]") ?? el, start: "top bottom", end: "bottom top", scrub: true } });
      });

      // drifting background shapes
      all("[data-drift]").forEach((el) => {
        gsap.to(el, { yPercent: Number(el.dataset.drift), ease: "none", scrollTrigger: { trigger: el.parentElement, start: "top bottom", end: "bottom top", scrub: true } });
      });

      // development timeline: line draws, nodes light up
      const line = scope.querySelector<HTMLElement>("[data-line]");
      const nodes = all("[data-node]");
      if (line && nodes.length) {
        nodes.forEach((n) => { n.dataset.on = "false"; });
        gsap.fromTo(line, wide ? { scaleX: 0 } : { scaleY: 0 }, {
          ...(wide ? { scaleX: 1 } : { scaleY: 1 }), ease: "none",
          scrollTrigger: {
            trigger: "[data-steps]", start: "top 72%", end: "bottom 62%", scrub: 0.5,
            onUpdate: (self) => nodes.forEach((n, i) => { n.dataset.on = self.progress >= (i / (nodes.length - 1)) * 0.98 ? "true" : "false"; }),
          },
        });
        cleanups.push(() => nodes.forEach((n) => { delete n.dataset.on; }));
      }

      requestAnimationFrame(() => ScrollTrigger.refresh());

      if (fine) {
        let tilt: HTMLElement | null = null;
        let mag: HTMLElement | null = null;
        const release = (el: HTMLElement) => gsap.to(el, { rotateX: 0, rotateY: 0, duration: 0.9, ease: SPRING, overwrite: "auto" });
        const unpull = (el: HTMLElement) => gsap.to(el, { x: 0, y: 0, duration: 0.8, ease: "elastic.out(1,.45)", overwrite: "auto" });
        const onMove = (e: PointerEvent) => {
          const target = e.target as Element;
          const card = target.closest<HTMLElement>("[data-tilt]");
          if (tilt && tilt !== card) release(tilt);
          tilt = card;
          if (card) {
            const b = card.getBoundingClientRect();
            card.style.setProperty("--mx", `${e.clientX - b.left}px`);
            card.style.setProperty("--my", `${e.clientY - b.top}px`);
            const x = (e.clientX - b.left) / b.width - 0.5;
            const y = (e.clientY - b.top) / b.height - 0.5;
            gsap.to(card, { rotateX: -y * 6, rotateY: x * 6, transformPerspective: 900, duration: 0.55, ease: "power3.out", overwrite: "auto" });
          }
          const m = target.closest<HTMLElement>("[data-mag]");
          if (mag && mag !== m) unpull(mag);
          mag = m;
          if (m) {
            const b = m.getBoundingClientRect();
            gsap.to(m, { x: (e.clientX - b.left - b.width / 2) * 0.28, y: (e.clientY - b.top - b.height / 2) * 0.28, duration: 0.5, ease: "power3.out", overwrite: "auto" });
          }
        };
        const onLeave = () => { if (tilt) release(tilt); if (mag) unpull(mag); tilt = null; mag = null; };
        scope.addEventListener("pointermove", onMove);
        scope.addEventListener("pointerleave", onLeave);
        cleanups.push(() => { scope.removeEventListener("pointermove", onMove); scope.removeEventListener("pointerleave", onLeave); });
      }

      return () => cleanups.forEach((fn) => fn());
    });

    return () => mm.revert();
  }, { scope: root });

  return <div ref={root}>{children}</div>;
}
