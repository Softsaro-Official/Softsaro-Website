"use client";

import { useGSAP } from "@gsap/react";
import { gsap } from "gsap";
import { ScrollTrigger } from "gsap/ScrollTrigger";
import { useRef } from "react";

export default function PricingMotion({ children }: { children: React.ReactNode }) {
  const root = useRef<HTMLDivElement>(null);
  useGSAP(() => {
    gsap.registerPlugin(ScrollTrigger);
    const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const scope = root.current;
    if (!scope || reduce) return;
    const hero = scope.querySelector<HTMLElement>(".price-hero");
    const heading = scope.querySelector<HTMLElement>(".price-heading");
    const cta = scope.querySelector<HTMLElement>(".price-cta");
    const cards = Array.from(scope.querySelectorAll<HTMLElement>(".price-card"));
    const offs: Array<() => void> = [];

    if (hero) {
      const eye = hero.querySelector(".price-eye");
      const title = hero.querySelector("h1");
      const copy = hero.querySelector("p:not(.price-eye)");
      const jump = hero.querySelector(".price-jump");
      gsap.timeline({ defaults: { ease: "expo.out" }, delay: .12 })
        .from(eye, { y: 20, autoAlpha: 0, duration: .7 })
        .from(title, { y: 72, rotate: 2, transformOrigin: "0% 100%", autoAlpha: 0, duration: 1.08 }, "-=.2")
        .from([copy, jump], { y: 24, autoAlpha: 0, duration: .72, stagger: .12 }, "+=.06");
      gsap.to(title, { yPercent: -6, ease: "none", scrollTrigger: { trigger: hero, start: "top top", end: "bottom top", scrub: true } });
    }

    if (heading) gsap.from(heading.children, { y: 28, autoAlpha: 0, duration: .8, stagger: .12, ease: "expo.out", scrollTrigger: { trigger: heading, start: "top 88%", once: true } });

    cards.forEach((card, index) => {
      const content = card.querySelectorAll(".price-card-top, h2, .price-value, small, ul, .price-feature-group, .price-extra, .price-action");
      gsap.fromTo(card, { y: 90, autoAlpha: 0, scale: .94, rotate: index === 1 ? 2 : -2 }, {
        y: 0, autoAlpha: 1, scale: 1, rotate: 0, duration: 1.15, ease: "expo.out",
        scrollTrigger: { trigger: card, start: "top 88%", once: true },
      });
      gsap.from(content, { y: 24, autoAlpha: 0, duration: .65, stagger: .045, ease: "power3.out", scrollTrigger: { trigger: card, start: "top 74%", once: true } });
      gsap.to(card, { yPercent: index === 1 ? -3 : 3, ease: "none", scrollTrigger: { trigger: card, start: "top bottom", end: "bottom top", scrub: true } });

      if (window.matchMedia("(hover: hover) and (pointer: fine)").matches) {
        const move = (event: PointerEvent) => {
          const box = card.getBoundingClientRect();
          const x = ((event.clientX - box.left) / box.width) * 100;
          const y = ((event.clientY - box.top) / box.height) * 100;
          card.style.setProperty("--px", `${x}%`);
          card.style.setProperty("--py", `${y}%`);
          gsap.to(card, { rotationY: (x - 50) * .035, rotationX: -(y - 50) * .025, transformPerspective: 1100, duration: .55, ease: "power3.out", overwrite: "auto" });
        };
        const leave = () => gsap.to(card, { rotationX: 0, rotationY: 0, duration: .9, ease: "elastic.out(1,.5)", overwrite: "auto" });
        card.addEventListener("pointermove", move);
        card.addEventListener("pointerleave", leave);
        offs.push(() => { card.removeEventListener("pointermove", move); card.removeEventListener("pointerleave", leave); });
      }
    });

    if (cta) gsap.from(cta.children, { y: 42, autoAlpha: 0, duration: .95, stagger: .12, ease: "expo.out", scrollTrigger: { trigger: cta, start: "top 82%", once: true } });
    const refresh = () => ScrollTrigger.refresh();
    window.addEventListener("load", refresh);
    document.fonts?.ready.then(refresh);
    return () => { window.removeEventListener("load", refresh); offs.forEach((off) => off()); };
  }, { scope: root });
  return <div ref={root}>{children}</div>;
}
