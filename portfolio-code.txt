/* project screenshots you add to public/images/portfolio/<slug>.(jpg|png|webp) are picked up automatically */
function localImages(): Record<string, string> {
  const out: Record<string, string> = {};
  try {
    fs.readdirSync(path.join(process.cwd(), "public", "images", "portfolio")).forEach((f) => {
      const m = f.match(/^(.+)\.(jpe?g|png|webp|avif)$/i);
      if (m) out[m[1].toLowerCase()] = `/images/portfolio/${f}`;
    });
  } catch { /* folder not created yet */ }
  return out;
}

function Cover({ name, cat, src, className = "", blur = false }: { name: string; cat: Cat; src?: string; className?: string; blur?: boolean }) {
  const [bad, setBad] = useState(false);
  if (src && !bad) return <>{blur && <span className="pf-bgblur" aria-hidden="true" style={{ backgroundImage: `url("${src}")` }} />}<img className={`pf-img ${className}`} src={src} alt={`${name} project`} loading="lazy" decoding="async" onError={() => setBad(true)} /></>;
  const initials = name.split(/\s+/).map((w) => w[0]).join("").slice(0, 2).toUpperCase();
  return (
    <div className={`pf-cover c-${cat} ${className}`} role="img" aria-label={`${name} project cover`}>
      <b>{initials}</b><span>{name}</span><i>{CAT_LABEL[cat]}</i>
    </div>
  );
}

function splitWords(el: HTMLElement): HTMLElement[] {
  const html = el.innerHTML; // keeps the accent <em>
  const out: HTMLElement[] = [];
  const walk = (n: Node) => Array.from(n.childNodes).forEach((c) => {
    if (c.nodeType === 3) {
      const f = document.createDocumentFragment();
      (c.textContent ?? "").split(/(\s+)/).forEach((w) => {
        if (!w.trim()) { f.append(w); return; }
        const wrap = document.createElement("span"); wrap.className = "pf-w";
        const i = document.createElement("span"); i.className = "pf-i"; i.textContent = w; wrap.append(i); f.append(wrap); out.push(i);
      });
      c.replaceWith(f);
    } else if (c.nodeType === 1) walk(c);
  });
  walk(el);
  el.setAttribute("aria-label", el.textContent ?? html);
  return out;
}
